```python
def _derive_billing_cycle_start(cycle_day: int) -> date:
    today = date.today()
    try:
        if today.day >= cycle_day:
            return today.replace(day=cycle_day)
        else:
            if today.month == 1:
                return today.replace(year=today.year - 1, month=12, day=cycle_day)
            else:
                return today.replace(month=today.month - 1, day=cycle_day)
    except ValueError:
        logger.warning("billing_cycle_start_day=%r caused date error — using day 1", cycle_day)
        return today.replace(day=1)
```

```python
def compute_pool_state(
    blocks: list,
    threshold_state,           # ThresholdState | None
    config_dir: Optional[Path] = None,
) -> PoolState:
    if config_dir is None:
        config_dir = _DEFAULT_CONFIG_DIR

    pool_size_usd, cycle_day = _read_pool_config(config_dir)

    # Determine billing cycle start — check cache first (D-06)
    cycle_start: date
    cached = _read_pool_spend_cache(config_dir)
    current_cycle_start = _derive_billing_cycle_start(cycle_day)
    if cached.get("billing_cycle_start"):
        try:
            cached_start = date.fromisoformat(cached["billing_cycle_start"])
            # Stale check (Pitfall 4): if cached start < current cycle start, ignore cache
            if cached_start >= current_cycle_start:
                cycle_start = cached_start
            else:
                cycle_start = current_cycle_start
        except (ValueError, TypeError):
            cycle_start = current_cycle_start
    else:
        cycle_start = current_cycle_start

    billing_cycle_start_str = cycle_start.isoformat()

    # Get threshold_tokens (None when calibrating)
    threshold_tokens: Optional[int] = None
    if threshold_state is not None:
        threshold_tokens = threshold_state.threshold_tokens

    # Sum OVERAGE session costs in billing period (D-01, D-02, D-03)
    pool_spend_usd = 0.0
    for block in blocks:
        # Skip active and gap blocks (only completed sessions count per D-01)
        if block.get("isActive", False) or block.get("isGap", False):
            continue
        # Skip sessions outside the billing period
        if not _in_billing_period(block.get("startTime", ""), cycle_start):
            continue
        # OVERAGE classification (D-01): totalTokens > threshold_tokens
        if _classify_overage(block, threshold_tokens):
            pool_spend_usd += block.get("costUSD", 0.0)

    pool_remaining_usd = max(0.0, pool_size_usd - pool_spend_usd)
    pool_pct_spent = (pool_spend_usd / pool_size_usd * 100) if pool_size_usd > 0 else 0.0

    # is_overage: True when current threshold is known AND pool_spend > 0
    # (i.e., at least one OVERAGE session exists in this billing period)
    is_overage = pool_spend_usd > 0.0 and threshold_tokens is not None

    result = PoolState(
        pool_size_usd=pool_size_usd,
        pool_spend_usd=pool_spend_usd,
        pool_remaining_usd=pool_remaining_usd,
        pool_pct_spent=pool_pct_spent,
        billing_cycle_start=billing_cycle_start_str,
        is_overage=is_overage,
    )

    # Write cache (D-03 / D-04): atomic write on every cycle
    _write_pool_spend_cache(config_dir, result)

    return result
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Tampering | pool_spend.json | accept | Local single-user tool; no remote write surface; stale-detection handles corrupted billing_cycle_start by falling back to derived current cycle start |
| T-03-02 | Tampering | config.json pool keys | mitigate | `_read_pool_config()` validates type and range: `pool_size_usd` must be positive (int|float), `billing_cycle_start_day` must be int 1–28; any other value logs warning and uses default — malformed input cannot crash the loop |
| T-03-03 | Denial of Service | _write_pool_spend_cache | mitigate | All write I/O wrapped in try/except; failure logs warning and continues — disk full or permission error does not halt monitoring |
| T-03-04 | Information Disclosure | pool_spend.json path | accept | File contains only estimated spend figures (local approximations); no auth tokens, no PII; single-user local tool; "est." prefix signals non-authoritative data (DISP-02) |
| T-03-05 | Elevation of Privilege | Path construction | mitigate | Config dir is `Path.home() / ".claude-monitor"` — no user-controlled path components; no path traversal surface in this module |
